20template <class T>
21static T integrate_dynamics_forward_euler(double dt, T x, T u, std::function<T(T, T)> dynamics) {
22 return x + dt * dynamics(x, u);
23}
24
27template <class T>
28static T integrate_dynamics_modified_euler(double dt, T x, T u, std::function<T(T, T)> dynamics) {
29 T k1 = dynamics(x, u);
30 T k2 = dynamics(x + dt * k1, u);
31
32 return x + dt * (k1 + k2) / 2;
33}
34
37template <class T>
38static T integrate_dynamics_rk4(double dt, T x, T u, std::function<T(T, T)> dynamics) {
39 T k1 = dynamics(x, u);
40 T k2 = dynamics(x + dt / 2 * k1, u);
41 T k3 = dynamics(x + dt / 2 * k2, u);
42 T k4 = dynamics(x + dt * k3, u);
43 return x + dt / 6 * (k1 + 2 * k2 + 2 * k3 + k4);
44}

103 Eigen::VectorXd dynamics_eval(Eigen::VectorXd x, Eigen::VectorXd u) {
104 casadi::DM x_dm = casadi::DM::zeros(nx(), 1);
105 casadi::DM u_dm = casadi::DM::zeros(nu(), 1);
106 for (size_t i = 0; i < nx(); i++) {
107 x_dm(i) = x[i];
108 }
109 for (size_t i = 0; i < nu(); i++) {
110 u_dm(i) = u[i];
111 }
112 casadi::MX dx_mx = dynamics(x_dm, u_dm);
113 casadi::DM dx_dm = casadi::MX::evalf(dx_mx);
114 Eigen::VectorXd dx = casadi_utils::to_eigen(dx_dm);
115 return dx;
116 }

120 Eigen::VectorXd simulate(Eigen::VectorXd x0, Eigen::MatrixXd u) {
121 assert(dyn_type_ == DynamicsType::Discretized);
122 return dynamics_eval(x0, u);

130 Eigen::VectorXd simulate(Eigen::VectorXd x0, Eigen::MatrixXd u, double dt) {
131 assert(dyn_type_ != DynamicsType::Discretized);
132 auto dyn =
133 std::bind(&Problem::dynamics_eval, this, std::placeholders::_1, std::placeholders::_2);
134 switch (dyn_type_) {
135 case DynamicsType::ContinuesForwardEuler:
136 return integrate_dynamics_forward_euler<Eigen::VectorXd>(dt, x0, u, dyn);
137 break;
138 case DynamicsType::ContinuesModifiedEuler:
139 return integrate_dynamics_modified_euler<Eigen::VectorXd>(dt, x0, u, dyn);
140 break;
141 case DynamicsType::ContinuesRK4:
142 return integrate_dynamics_rk4<Eigen::VectorXd>(dt, x0, u, dyn);
143 break;
144 case DynamicsType::Discretized:
145 break;
146 }
147 return x0;
148 }

401 template <class T>
402 MPC(std::shared_ptr<T> prob, std::string solver_name = "ipopt",
403 casadi::Dict config = default_ipopt_config())
404 : prob_(prob), solver_name_(solver_name), config_(config) {
405 using namespace casadi;
406 static_assert(std::is_base_of_v<Problem, T>, "prob must be based SimpleProb");
407
408 const size_t nx = prob_->nx();
409 const size_t nu = prob_->nu();
410 const size_t N = prob_->horizon();
411
412 // mapsum_stage_cost: build the stage-cost sum as a MapSum so first-order AD
413 // stays loop-shaped (smaller derivative graph).
414 // expand_inner_functions: SX-expand per-stage F/L/G before .map(N) so the
415 // inner per-stage AD operates on flat SX.
416 bool mapsum_stage_cost = true;
417 bool expand_inner_functions = true;
418 {
419 auto it = config_.find("mapsum_stage_cost");
420 if (it != config_.end()) {
421 mapsum_stage_cost = static_cast<bool>(it->second);
422 config_.erase(it);
423 }
424 it = config_.find("expand_inner_functions");
425 if (it != config_.end()) {
426 expand_inner_functions = static_cast<bool>(it->second);
427 config_.erase(it);
428 }
429 }
430
431 build_with_map(nx, nu, N, mapsum_stage_cost, expand_inner_functions);
432
433 if (expand_inner_functions) {
434 config_["expand"] = false;
435 }
436
437 if (equality_required(solver_name_, config_)) {
438 // Convert std::vector<bool> to std::vector<casadi_int> for CasADi
439 std::vector<casadi_int> equality_int(equality_.begin(), equality_.end());
440 config_["equality"] = equality_int;
441 }
442
443 build_solver();
444 }

456 virtual Eigen::VectorXd solve(Eigen::VectorXd x0,
457 casadi::DMDict new_param_list = casadi::DMDict()) {
458 using namespace casadi;
459
460 // Set new parameter
461 for (auto &[param_name, param] : new_param_list) {
462 prob_->param_list_[param_name].dm = param;
463 }
464
465 const size_t nx = prob_->nx();
466 const size_t nu = prob_->nu();
467
468 for (size_t l = 0; l < nx; l++) {
469 lbw_(l) = x0[l];
470 ubw_(l) = x0[l];
471 }
472
473 DMDict arg;
474 arg["x0"] = w0_;
475 arg["lbx"] = lbw_;
476 arg["ubx"] = ubw_;
477 arg["lbg"] = lbg_;
478 arg["ubg"] = ubg_;
479 arg["lam_x0"] = lam_x0_;
480 arg["lam_g0"] = lam_g0_;
481 param_vec_.clear();
482 param_vec_.reserve(prob_->param_list_.size());
483 for (auto &[param_name, param_pair] : prob_->param_list_) {
484 param_vec_.push_back(param_pair.dm);
485 }
486 arg["p"] = vertcat(param_vec_);
487 DMDict sol = solver_(arg);
488
489 w0_ = sol["x"];
490 lam_x0_ = sol["lam_x"];
491 lam_g0_ = sol["lam_g"];
492
493 Eigen::VectorXd opt_u(nu);
494 std::copy(w0_.ptr() + nx, w0_.ptr() + nx + nu, opt_u.data());
495
496 return opt_u;
497 }

531 // Build NLP with map (for expand=false, faster JIT compilation)
532 void build_with_map(size_t nx, size_t nu, casadi_int N, bool mapsum_stage_cost = true,
533 bool expand_inner_functions = true) {
534 using namespace casadi;
535 double inf = std::numeric_limits<double>::infinity(); // Make sure inf is defined
536
537 // 1. Symbolic variables - create individual variables for each stage
538 Xs.reserve(N + 1);
539 Us.reserve(N);
540 for (casadi_int i = 0; i < N; i++) {
541 Xs.push_back(MX::sym("X_" + std::to_string(i), nx, 1));
542 Us.push_back(MX::sym("U_" + std::to_string(i), nu, 1));
543 }
544 Xs.push_back(MX::sym("X_" + std::to_string(N), nx, 1));
545
546 // Create matrices for map operations
547 MX X = horzcat(Xs);
548 MX U = horzcat(Us);
549
550 MX x_k = MX::sym("x_k", nx);
551 MX u_k = MX::sym("u_k", nu);
552
553 // Collect all parameters
554 std::vector<MX> params_mx;
555 for (auto &[param_name, param_pair] : prob_->param_list_)
556 params_mx.push_back(param_pair.mx);
557
558 // 2. CasADi Functions for one step (unchanged)
559 // std::function<MX(MX, MX)> dynamics_func;
560 // ... same as your code ...
561 MX x_next;
562 switch (prob_->dynamics_type()) {
563 case Problem::DynamicsType::ContinuesForwardEuler: {
564 std::function<casadi::MX(casadi::MX, casadi::MX)> con_dyn =
565 std::bind(&Problem::dynamics, prob_.get(), std::placeholders::_1, std::placeholders::_2);
566 x_next = integrate_dynamics_forward_euler<casadi::MX>(prob_->dt(), x_k, u_k, con_dyn);
567 break;
568 }
569 case Problem::DynamicsType::ContinuesModifiedEuler: {
570 std::function<casadi::MX(casadi::MX, casadi::MX)> con_dyn =
571 std::bind(&Problem::dynamics, prob_.get(), std::placeholders::_1, std::placeholders::_2);
572 x_next = integrate_dynamics_modified_euler<casadi::MX>(prob_->dt(), x_k, u_k, con_dyn);
573 break;
574 }
575 case Problem::DynamicsType::ContinuesRK4: {
576 std::function<casadi::MX(casadi::MX, casadi::MX)> con_dyn =
577 std::bind(&Problem::dynamics, prob_.get(), std::placeholders::_1, std::placeholders::_2);
578 x_next = integrate_dynamics_rk4<casadi::MX>(prob_->dt(), x_k, u_k, con_dyn);
579 break;
580 }
581 case Problem::DynamicsType::Discretized:
582 x_next = prob_->dynamics(x_k, u_k);
583 break;
584 }
585 Function F("F_dynamics", {x_k, u_k}, {x_next});
586 if (expand_inner_functions)
587 F = F.expand(F.name(), {{"cse", true}});
588
589 std::vector<MX> L_inputs = {x_k, u_k};
590 L_inputs.insert(L_inputs.end(), params_mx.begin(), params_mx.end());
591 MX stage_cost = prob_->stage_cost(x_k, u_k, 0);
592 Function L("L_stage_cost", L_inputs, {stage_cost});
593 if (expand_inner_functions)
594 L = L.expand(L.name(), {{"cse", true}});
595
596 // 制約一覧
597 std::vector<MX> g_k_vec;
598 for (auto &con : prob_->equality_constrinats_) {
599 g_k_vec.push_back(con(x_k, u_k));
600 }
601 for (auto &con : prob_->inequality_constrinats_) {
602 g_k_vec.push_back(con(x_k, u_k));
603 }
604
605 // ソフト制約用のスラック変数を per-stage に確保し、対応する不等式を g_k に追加する。
606 // - 不等式 g(x,u) <= 0 -> g - s <= 0
607 // - 等式 h(x,u) = 0 -> h - s <= 0 かつ -h - s <= 0 (|h| <= s)
608 // s >= 0 は w 側の bound で強制し、ペナルティ w1*1^T s + 0.5*w2*s^T s をコストに加算する。
609 std::vector<casadi_int> soft_sizes;
610 soft_sizes.reserve(prob_->soft_constraints_.size());
611 for (auto &sc : prob_->soft_constraints_) {
612 soft_sizes.push_back(sc.func(x_k, u_k).size1());
613 }
614 const casadi_int n_s_total =
615 std::accumulate(soft_sizes.begin(), soft_sizes.end(), static_cast<casadi_int>(0));
616
617 std::vector<MX> Ss;
618 MX s_k;
619 if (n_s_total > 0) {
620 s_k = MX::sym("s_k", n_s_total, 1);
621 Ss.reserve(N);
622 for (casadi_int i = 0; i < N; ++i) {
623 Ss.push_back(MX::sym("S_" + std::to_string(i), n_s_total, 1));
624 }
625
626 casadi_int s_offset = 0;
627 for (size_t i = 0; i < prob_->soft_constraints_.size(); ++i) {
628 auto &sc = prob_->soft_constraints_[i];
629 const casadi_int m = soft_sizes[i];
630 MX s_part = s_k(Slice(s_offset, s_offset + m));
631 MX c_val = sc.func(x_k, u_k);
632 if (sc.type == Problem::ConstraintType::Inequality) {
633 g_k_vec.push_back(c_val - s_part);
634 } else {
635 g_k_vec.push_back(c_val - s_part);
636 g_k_vec.push_back(-c_val - s_part);
637 }
638 s_offset += m;
639 }
640 }
641
642 MX g_k = vertcat(g_k_vec);
643
644 std::vector<MX> G_inputs = {x_k, u_k};
645 if (n_s_total > 0) {
646 G_inputs.push_back(s_k);
647 }
648 G_inputs.insert(G_inputs.end(), params_mx.begin(), params_mx.end());
649 Function G_constraints("G_constraints", G_inputs, {g_k});
650 if (expand_inner_functions)
651 G_constraints = G_constraints.expand(G_constraints.name(), {{"cse", true}});
652
653 // 3. Map application
654 MX X_next_cal = F.map(N)(std::vector<MX>{X(Slice(), Slice(0, N)), U})[0];
655
656 // Stage cost: when mapsum_stage_cost is set, replace each per-stage param
657 // p (shape rows×N) by repmat(col_sym, 1, N) inside the user's stage_cost
658 // so p(:, k) collapses to a single column symbol; build the cost via
659 // MapSum. Verify k-independence by comparing the substituted expression at
660 // every k against k=0; on mismatch (k-dependent branching beyond param
661 // slicing), fall back to the per-stage loop.
662 bool mapsum_safe = mapsum_stage_cost;
663 std::vector<MX> col_syms;
664 std::vector<size_t> per_stage_idx;
665 MX cost_substituted;
666
667 if (mapsum_safe) {
668 for (size_t p = 0; p < params_mx.size(); ++p) {
669 if (params_mx[p].size2() == N) {
670 per_stage_idx.push_back(p);
671 col_syms.push_back(MX::sym(params_mx[p].name() + "_col", params_mx[p].size1(), 1));
672 }
673 }
674
675 std::vector<MX> per_stage_orig;
676 std::vector<MX> per_stage_magic;
677 per_stage_orig.reserve(per_stage_idx.size());
678 per_stage_magic.reserve(per_stage_idx.size());
679 for (size_t i = 0; i < per_stage_idx.size(); ++i) {
680 per_stage_orig.push_back(params_mx[per_stage_idx[i]]);
681 per_stage_magic.push_back(repmat(col_syms[i], 1, N));
682 }
683 cost_substituted =
684 per_stage_orig.empty()
685 ? stage_cost
686 : MX::substitute(std::vector<MX>{stage_cost}, per_stage_orig, per_stage_magic)[0];
687
688 for (casadi_int k = 1; k < N; ++k) {
689 MX cost_at_k = prob_->stage_cost(x_k, u_k, k);
690 MX cost_k_subst =
691 per_stage_orig.empty()
692 ? cost_at_k
693 : MX::substitute(std::vector<MX>{cost_at_k}, per_stage_orig, per_stage_magic)[0];
694 if (!MX::is_equal(cost_substituted, cost_k_subst, /*depth*/ 100)) {
695 casadi_warning("mapsum_stage_cost requested but stage_cost depends on k "
696 "beyond per-stage parameter slicing; falling back to "
697 "per-stage loop to preserve correctness.");
698 mapsum_safe = false;
699 break;
700 }
701 }
702 }
703
704 MX J_stage;
705 if (mapsum_safe) {
706 // Per-stage params -> col_syms (Map iterates columns); broadcast params
707 // are marked via reduce_in (Map repeats them).
708 std::vector<MX> L_subst_inputs = {x_k, u_k};
709 std::vector<casadi_int> reduce_in;
710 size_t next_col = 0;
711 for (size_t p = 0; p < params_mx.size(); ++p) {
712 casadi_int input_idx = static_cast<casadi_int>(2 + p);
713 bool is_per_stage = (next_col < per_stage_idx.size() && per_stage_idx[next_col] == p);
714 if (is_per_stage) {
715 L_subst_inputs.push_back(col_syms[next_col]);
716 ++next_col;
717 } else {
718 L_subst_inputs.push_back(params_mx[p]);
719 reduce_in.push_back(input_idx);
720 }
721 }
722 Function L_subst("L_stage_subst", L_subst_inputs, {cost_substituted});
723 if (expand_inner_functions)
724 L_subst = L_subst.expand(L_subst.name(), {{"cse", true}});
725
726 std::vector<casadi_int> reduce_out = {0};
727 Function L_mapsum = L_subst.map("L_stage_mapsum", "serial", N, reduce_in, reduce_out, Dict());
728 std::vector<MX> map_inputs = {X(Slice(), Slice(0, N)), U};
729 for (auto &param : params_mx) {
730 map_inputs.push_back(param);
731 }
732 J_stage = L_mapsum(map_inputs)[0];
733 } else {
734 std::vector<MX> stage_costs;
735 stage_costs.reserve(N);
736 for (casadi_int i = 0; i < N; ++i) {
737 std::vector<MX> stage_cost_inputs = {Xs[i], Us[i]};
738 stage_cost_inputs.insert(stage_cost_inputs.end(), params_mx.begin(), params_mx.end());
739 MX cost_i = prob_->stage_cost(Xs[i], Us[i], i);
740 Function L_i("L_stage_cost_" + std::to_string(i), stage_cost_inputs, {cost_i});
741 stage_costs.push_back(L_i(stage_cost_inputs)[0]);
742 }
743 J_stage = sum(vertcat(stage_costs));
744 }
745
746 // Terminal cost
747 MX terminal_val = prob_->terminal_cost(Xs[N]);
748
749 // ソフト制約のスラック変数に対するペナルティコスト
750 MX penalty_cost = 0;
751 if (n_s_total > 0) {
752 for (casadi_int k = 0; k < N; ++k) {
753 casadi_int off = 0;
754 for (size_t i = 0; i < prob_->soft_constraints_.size(); ++i) {
755 auto &sc = prob_->soft_constraints_[i];
756 const casadi_int m = soft_sizes[i];
757 MX sk_part = Ss[k](Slice(off, off + m));
758 penalty_cost = penalty_cost + sc.w1 * sum1(sk_part);
759 if (sc.w2 != 0.0) {
760 penalty_cost = penalty_cost + 0.5 * sc.w2 * dot(sk_part, sk_part);
761 }
762 off += m;
763 }
764 }
765 }
766
767 MX J = J_stage + terminal_val + penalty_cost;
768
769 // Path constraints
770 MX G_path;
771 if (!g_k.is_empty()) {
772 std::vector<MX> G_map_inputs = {X(Slice(), Slice(0, N)), U};
773 if (n_s_total > 0) {
774 G_map_inputs.push_back(horzcat(Ss));
775 }
776 for (auto &param : params_mx) {
777 G_map_inputs.push_back(repmat(param, 1, N));
778 }
779 G_path = G_constraints.map(N)(G_map_inputs)[0];
780 }
781
782 // 4. NLP construction
783 std::vector<MX> w_vec;
784 w_vec.reserve((n_s_total > 0 ? 3 : 2) * N + 1);
785 for (casadi_int i = 0; i < N; ++i) {
786 w_vec.push_back(Xs[i]);
787 w_vec.push_back(Us[i]);
788 if (n_s_total > 0) {
789 w_vec.push_back(Ss[i]);
790 }
791 }
792 w_vec.push_back(Xs[N]);
793 MX w = vertcat(w_vec);
794
795 std::vector<MX> g_vec;
796 g_vec.push_back(reshape(X(Slice(), Slice(1, N + 1)) - X_next_cal, nx * N, 1));
797 if (!g_k.is_empty()) {
798 g_vec.push_back(reshape(G_path, G_path.size1() * G_path.size2(), 1));
799 }
800
801 // --- [FIX] Build bounds in temporary double vectors first ---
802 std::vector<double> lbw_numeric, ubw_numeric, lbg_numeric, ubg_numeric;
803
804 auto &u_bounds = prob_->u_bounds_;
805 auto &x_bounds = prob_->x_bounds_;
806
807 // Bounds for w
808 for (casadi_int i = 0; i < N; ++i) {
809 if (i == 0) { // Dummy bounds for x_0 (will be overwritten by x0)
810 lbw_numeric.insert(lbw_numeric.end(), nx, 0.0);
811 ubw_numeric.insert(ubw_numeric.end(), nx, 0.0);
812 } else {
813 lbw_numeric.insert(lbw_numeric.end(), x_bounds[i - 1].first.data(),
814 x_bounds[i - 1].first.data() + nx);
815 ubw_numeric.insert(ubw_numeric.end(), x_bounds[i - 1].second.data(),
816 x_bounds[i - 1].second.data() + nx);
817 }
818 lbw_numeric.insert(lbw_numeric.end(), u_bounds[i].first.data(),
819 u_bounds[i].first.data() + nu);
820 ubw_numeric.insert(ubw_numeric.end(), u_bounds[i].second.data(),
821 u_bounds[i].second.data() + nu);
822 if (n_s_total > 0) {
823 lbw_numeric.insert(lbw_numeric.end(), n_s_total, 0.0);
824 ubw_numeric.insert(ubw_numeric.end(), n_s_total, inf);
825 }
826 }
827 // Bounds for x_N
828 lbw_numeric.insert(lbw_numeric.end(), x_bounds[N - 1].first.data(),
829 x_bounds[N - 1].first.data() + nx);
830 ubw_numeric.insert(ubw_numeric.end(), x_bounds[N - 1].second.data(),
831 x_bounds[N - 1].second.data() + nx);
832
833 // Bounds for g
834 // Continuity constraints are all zero
835 lbg_numeric.insert(lbg_numeric.end(), nx * N, 0.0);
836 ubg_numeric.insert(ubg_numeric.end(), nx * N, 0.0);
837 equality_.insert(equality_.end(), nx * N, true);
838
839 // Path constraints bounds
840 for (casadi_int i = 0; i < N; ++i) {
841 for (auto &con : prob_->equality_constrinats_) {
842 auto con_val = con(x_k, u_k); // For getting size
843 lbg_numeric.insert(lbg_numeric.end(), con_val.size1(), 0.0);
844 ubg_numeric.insert(ubg_numeric.end(), con_val.size1(), 0.0);
845
846 equality_.insert(equality_.end(), con_val.size1(), true);
847 }
848 for (auto &con : prob_->inequality_constrinats_) {
849 auto con_val = con(x_k, u_k); // For getting size
850 lbg_numeric.insert(lbg_numeric.end(), con_val.size1(), -inf);
851 ubg_numeric.insert(ubg_numeric.end(), con_val.size1(), 0.0);
852
853 equality_.insert(equality_.end(), con_val.size1(), false);
854 }
855 // ソフト制約の path 制約 (どちらも片側不等式 [-inf, 0])
856 for (size_t s_i = 0; s_i < prob_->soft_constraints_.size(); ++s_i) {
857 const casadi_int m = soft_sizes[s_i];
858 const casadi_int rows =
859 (prob_->soft_constraints_[s_i].type == Problem::ConstraintType::Inequality) ? m : 2 * m;
860 lbg_numeric.insert(lbg_numeric.end(), rows, -inf);
861 ubg_numeric.insert(ubg_numeric.end(), rows, 0.0);
862 equality_.insert(equality_.end(), rows, false);
863 }
864 }
865
866 // Assign from the temporary numeric vectors
867 lbw_ = casadi::DM(lbw_numeric);
868 ubw_ = casadi::DM(ubw_numeric);
869 lbg_ = casadi::DM(lbg_numeric);
870 ubg_ = casadi::DM(ubg_numeric);
871
872 MX g_all = vertcat(g_vec);
873 MX p_all = vertcat(params_mx);
874
875 casadi_prob_ = {{"x", w}, {"f", J}, {"g", g_all}, {"p", p_all}};
876
877 // Initialize w0_, lam_x0_, lam_g0_ for warm start
878 w0_ = DM::zeros(w.size1(), 1);
879 lam_x0_ = DM::zeros(w.size1(), 1);
880 lam_g0_ = DM::zeros(vertcat(g_vec).size1(), 1);
881 }
882
883 virtual void build_solver() { solver_ = nlpsol("solver", solver_name_, casadi_prob_, config_); }

885private:
886};
